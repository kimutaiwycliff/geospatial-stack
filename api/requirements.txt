# Observability (brief §12). prometheus-fastapi-instrumentator over hand-rolled
# middleware -- it gives per-route request count/latency histograms (what §12
# actually asks for) out of the box, correctly excludes /metrics from its own
# counters, and is one line to wire (app/main.py). sentry-sdk's FastAPI +
# Celery integrations both no-op safely with no DSN set (see app/main.py /

# do anything, error or otherwise, until SENTRY_DSN is actually set.
prometheus-fastapi-instrumentator
sentry-sdk[fastapi,celery]

# FastAPI-native auth (brief §10): pyjwt for HS256 access tokens,
# passlib[bcrypt] for password hashing. Deliberately not fastapi-users --
# the whole auth surface here is "one admin role, one login endpoint, no
# signup" (see app/auth/security.py), which fastapi-users' user-management
# machinery would be overkill for.
pyjwt==2.9.0
passlib==1.7.4
# Pinned below passlib 1.7.4's own `bcrypt` extra: bcrypt>=4.1 dropped the
# `__about__` attribute passlib 1.7.4 probes for its version check, and
# newer bcrypt's stricter 72-byte-secret handling breaks passlib's internal
# self-test (raises instead of truncating) -- both surface as every
# hash/verify call failing. 4.0.1 is the last release passlib 1.7.4 works
# with cleanly; revisit together if passlib ever ships a fix.
bcrypt==4.0.1
# FastAPI's OAuth2PasswordRequestForm (app/routers/auth.py's /auth/login)

python-multipart==0.0.12
